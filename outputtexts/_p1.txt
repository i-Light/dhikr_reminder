            }
            if (_announced.add(index)) {
              debugPrint('PROBE: $index ${_phases[index]}');
            }
            return Center(
              child: SizedBox(
                width: 520,
                height: 320,
                child: _ProbeCase(index: index, t: _ticker.value),
              ),
            );
          },
        ),
      ),
    );
  }
}

  final double t;
